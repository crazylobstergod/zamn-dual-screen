cmake_minimum_required(VERSION 3.22.1)
project(snes_frontend)
set(CORE ${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/snes9x-libretro)
file(GLOB CORE_C ${CORE}/*.c ${CORE}/filter/*.c ${CORE}/libretro/libretro-common/compat/*.c ${CORE}/libretro/libretro-common/encodings/*.c ${CORE}/libretro/libretro-common/file/*.c ${CORE}/libretro/libretro-common/streams/*.c ${CORE}/libretro/libretro-common/string/*.c ${CORE}/libretro/libretro-common/time/*.c ${CORE}/libretro/libretro-common/vfs/*.c)
file(GLOB CORE_CPP ${CORE}/*.cpp ${CORE}/apu/apu.cpp)
list(APPEND CORE_CPP ${CORE}/apu/bapu/dsp/sdsp.cpp ${CORE}/apu/bapu/smp/smp.cpp ${CORE}/apu/bapu/smp/smp_state.cpp ${CORE}/libretro/libretro.cpp)
add_library(snes9x_core STATIC ${CORE_C} ${CORE_CPP})
target_include_directories(snes9x_core PRIVATE ${CORE} ${CORE}/libretro ${CORE}/apu ${CORE}/libretro/libretro-common/include)
target_compile_definitions(snes9x_core PRIVATE __ANDROID__ HAVE_STRINGS_H=1)
target_compile_options(snes9x_core PRIVATE -fPIC -fno-exceptions -fno-rtti)
add_library(snes_frontend SHARED frontend.cpp)
target_include_directories(snes_frontend PRIVATE ${CORE} ${CORE}/libretro)
target_link_libraries(snes_frontend PRIVATE -Wl,--whole-archive snes9x_core -Wl,--no-whole-archive android jnigraphics log)
